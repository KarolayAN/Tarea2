def gaussxw(N):
    """Calculates the points and weights for the Gauss-Legendre quadrature.

    Examples:
        >>> x, w = gaussxw(3)
        >>> len(x)
        3
        >>> len(w)
        3

    Args:
        N (int): Number of quadrature points.

    Returns:
        tuple: Two NumPy arrays: `x` (points) and `w` (weights).
    """
    x, w = np.polynomial.legendre.leggauss(N)
    return x, w

def gaussxwab(a, b, x, w):
    """Transforms the points from the interval [-1, 1] to the interval [a, b].

    Examples:
        >>> x = np.array([-1, 0, 1])
        >>> w = np.array([1, 1, 1])
        >>> gaussxwab(0, np.pi, x, w)
        (array([0.        , 1.57079633, 3.14159265]), array([1.57079633, 1.57079633, 1.57079633]))

    Args:
        x (ndarray): Points in the interval [-1, 1].
        a (float): Lower limit of the integration interval.
        b (float): Upper limit of the integration interval.

    Returns:
        tuple: Two arrays, transformed points `xp` and adjusted weights `wp`.
    """
    xp = 0.5 * (b - a) * x + 0.5 * (b + a)
    wp = 0.5 * (b - a) * w
    return xp, wp

def integrar(a, b, N):
    """Calculates the integral of `f` over the interval [a, b] using Gauss-Legendre quadrature.

    This function uses the Gauss-Legendre quadrature method to approximate the integral 
    of the function `sin(x^2)` over the interval [a, b].

    Examples:
        >>> f = lambda x: np.sin(x**2)
        >>> integrar(0, np.pi, 10)
        1.253...

    Args:
        a (float): Lower limit of the integration interval.
        b (float): Upper limit of the integration interval.
        N (int): Number of quadrature points.

    Returns:
        float: The result of the integral approximation.
    """
    x, w = gaussxw(N)  # Get the points and weights of Gauss-Legendre
    xp, wp = gaussxwab(a, b, x, w)  # Transform the points to the interval [a, b]
    return np.sum(wp * np.sin(xp**2))  # Calculate the integral approximation

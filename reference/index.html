def gauss_integrate(f, a, b, N):
    """Calcula la integral de la función `f` en el intervalo [a, b] usando cuadratura Gauss-Legendre.

    La cuadratura Gaussiana permite obtener una aproximación precisa de la integral
    de una función utilizando los puntos de Legendre y los pesos calculados con el
    método `gaussxw`.

    Args:
        f (function): Función a integrar.
        a (float): Límite inferior del intervalo de integración.
        b (float): Límite superior del intervalo de integración.
        N (int): Número de puntos de cuadratura.

    Returns:
        float: Resultado de la integral aproximada.

    Examples:
        >>> import numpy as np
        >>> f = lambda x: np.sin(x**2)
        >>> gauss_integrate(f, 0, np.pi, 10)
        1.253...
    """
    # Obtener los puntos y pesos de Gauss-Legendre
    x, w = gaussxw(N)

    # Transformar los puntos al intervalo [a, b]
    xp = transform(x, a, b)

    # Ajustar los pesos para el intervalo [a, b]
    wp = 0.5 * (b - a) * w

    # Calcular la suma ponderada para obtener la integral
    return np.sum(wp * f(xp))

def gaussxw(N):
    """Calcula los puntos y pesos de la cuadratura Gauss-Legendre.

    Este método se utiliza para obtener los puntos de muestreo y los pesos
    necesarios para aplicar la cuadratura Gaussiana sobre un intervalo.

    Args:
        N (int): Número de puntos de cuadratura.

    Returns:
        tuple: Dos arreglos de NumPy: puntos `x` y pesos `w`.

    Examples:
        >>> x, w = gaussxw(3)
        >>> len(x)
        3
        >>> len(w)
        3
    """
    a = np.linspace(3, 4*N - 1, N) / (4*N + 2)
    x = np.cos(np.pi * a)
    eps = 1e-15
    delta = 1.0
    while np.max(np.abs(delta)) > eps:
        p0 = np.ones(N)
        p1 = x.copy()
        for k in range(1, N):
            p0, p1 = p1, ((2*k + 1)*x*p1 - k*p0)/(k + 1)
        dp = N*(x*p1 - p0)/(x**2 - 1)
        delta = p1/dp
        x -= delta
    w = 2 / ((1 - x**2) * dp**2)
    return x, w

def transform(x, a, b):
    """Transforma los puntos del intervalo [-1, 1] al intervalo [a, b].

    Este método se utiliza para reescalar los puntos obtenidos con la cuadratura 
    Gaussiana del intervalo estándar [-1, 1] al intervalo [a, b] deseado.

    Args:
        x (ndarray): Puntos en el intervalo [-1, 1].
        a (float): Límite inferior del intervalo de integración.
        b (float): Límite superior del intervalo de integración.

    Returns:
        ndarray: Puntos transformados al intervalo [a, b].
    """
    return 0.5 * (b - a) * x + 0.5 * (b + a)
